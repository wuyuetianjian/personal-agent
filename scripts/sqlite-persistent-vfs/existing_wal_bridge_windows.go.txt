//go:build windows && nativeprobe_sqlite_persistent && nativeprobe_sqlite_main_read && nativeprobe_sqlite_existing_wal

// Existing-only WAL authority experiment. No stock xOpen forwarding or SHM.
package sqlite3

import (
	"fmt"
	"modernc.org/libc"
	"sync"
	"sync/atomic"
	"unsafe"
)

type ExperimentalExistingWALAuthority interface {
	ExperimentalMainReadAuthority
	OpenWALReadOnly() (ExperimentalMainReadLease, error)
}

var experimentalExistingWALMu sync.Mutex
var experimentalExistingWALPending = map[uintptr]*atomic.Uint64{}

func ExperimentalExistingWALRegisterRO(a ExperimentalExistingWALAuthority, canonical string) (uintptr, error) {
	if a == nil {
		return 0, fmt.Errorf("missing readonly existing authority")
	}
	return experimentalMainReadRegisterPrepared(a, canonical, __ccgo_fp(experimentalExistingWALOpen), func(v uintptr) {
		experimentalExistingWALMu.Lock()
		experimentalExistingWALPending[v] = &atomic.Uint64{}
		experimentalExistingWALMu.Unlock()
	})
}
func ExperimentalExistingWALPendingHits(v uintptr) uint64 {
	experimentalExistingWALMu.Lock()
	n := experimentalExistingWALPending[v]
	experimentalExistingWALMu.Unlock()
	if n == nil {
		return 0
	}
	return n.Load()
}
func experimentalExistingWALOpen(tls *libc.TLS, v, name, file uintptr, flags int32, out uintptr) int32 {
	if out != 0 {
		*(*int32)(unsafe.Pointer(out)) = 0
	}
	b := experimentalMainReadBindingFor(v)
	if b == nil || name == 0 || file == 0 {
		return SQLITE_CANTOPEN
	}
	if libc.GoString(name) == b.canonical && flags == SQLITE_OPEN_READONLY|SQLITE_OPEN_MAIN_DB {
		return experimentalMainReadOpen(tls, v, name, file, flags, out)
	}
	if libc.GoString(name) != b.canonical+"-wal" || flags != SQLITE_OPEN_READWRITE|SQLITE_OPEN_CREATE|SQLITE_OPEN_WAL {
		return SQLITE_CANTOPEN
	}
	experimentalExistingWALMu.Lock()
	n := experimentalExistingWALPending[v]
	experimentalExistingWALMu.Unlock()
	if n == nil {
		return SQLITE_CANTOPEN
	}
	n.Add(1)
	// Native RED boundary. Do not acquire/initialize/publish, and do not call stock.
	return SQLITE_CANTOPEN
}
