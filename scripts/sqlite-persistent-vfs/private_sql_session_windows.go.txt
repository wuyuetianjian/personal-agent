package sqlite3

import (
	"fmt"
	"modernc.org/libc"
	"sync"
	"sync/atomic"
	"unsafe"
)

type ExperimentalPrivateSQLAuthority interface {
	ExperimentalSQLAdmissionAuthority
	ExperimentalSHMSlotAuthority
	ExperimentalPrivateSQLStorageWitness
}

var experimentalPrivateSQLMu sync.Mutex
var experimentalPrivateSQLRefusals = map[uintptr]*atomic.Uint64{}

func ExperimentalPrivateSQLSessionOpenRefusals(v uintptr) uint64 {
	experimentalPrivateSQLMu.Lock()
	p := experimentalPrivateSQLRefusals[v]
	experimentalPrivateSQLMu.Unlock()
	if p == nil {
		return 0
	}
	return p.Load()
}
func ExperimentalPrivateSQLSessionRegister(a ExperimentalPrivateSQLAuthority, canonical string) (uintptr, string, error) {
	if a == nil || canonical == "" {
		return 0, "", fmt.Errorf("missing private SQL authority")
	}
	admission := &experimentalSQLAdmissionBinding{authority: a, canonical: canonical}
	v, e := experimentalMainReadRegisterPrepared(a, canonical, __ccgo_fp(experimentalPrivateSQLSessionOpen), func(v uintptr) {
		p := (*Tsqlite3_vfs)(unsafe.Pointer(v))
		p.FxAccess = __ccgo_fp(experimentalSQLAdmissionAccess)
		p.FxFullPathname = __ccgo_fp(experimentalSQLAdmissionFullPath)
		experimentalSQLAdmissionMu.Lock()
		experimentalSQLAdmissionBindings[v] = admission
		experimentalSQLAdmissionMu.Unlock()
		experimentalExistingWALMu.Lock()
		experimentalExistingWALPending[v] = &atomic.Uint64{}
		experimentalExistingWALMu.Unlock()
		experimentalSHMResourceMu.Lock()
		experimentalSHMResourceBindings[v] = &experimentalSHMResourceBinding{authority: a}
		experimentalSHMResourceMu.Unlock()
		experimentalSHMSlotMu.Lock()
		experimentalSHMSlotBindings[v] = &experimentalSHMSlotBinding{authority: a}
		experimentalSHMSlotMu.Unlock()
		experimentalPrivateSQLMu.Lock()
		experimentalPrivateSQLRefusals[v] = &atomic.Uint64{}
		experimentalPrivateSQLMu.Unlock()
	})
	if e != nil {
		admission.disabled.Store(true)
		return 0, "", e
	}
	return v, libc.GoString((*Tsqlite3_vfs)(unsafe.Pointer(v)).FzName), nil
}

// All tables and bindings are prepared before registration. Only the fixed,
// internally generated URI is admitted; this is not an arbitrary URI gateway.
func experimentalPrivateSQLSessionOpen(tls *libc.TLS, v, name, file uintptr, flags int32, out uintptr) (rc int32) {
	rc = SQLITE_CANTOPEN
	if out != 0 {
		*(*int32)(unsafe.Pointer(out)) = 0
	}
	if file != 0 {
		(*Tsqlite3_file)(unsafe.Pointer(file)).FpMethods = 0
	}
	b := experimentalSQLAdmissionLookup(v)
	if b == nil || b.disabled.Load() || name == 0 || file == 0 || out == 0 {
		return
	}
	defer func() {
		if p := recover(); p != nil {
			if binding := experimentalMainReadBindingFor(v); binding != nil {
				binding.poison(experimentalMainReadPanic("SQL session open", p))
			}
			*(*int32)(unsafe.Pointer(out)) = 0
			rc = SQLITE_CANTOPEN
		}
	}()
	if libc.GoString(name) == b.canonical {
		if flags != SQLITE_OPEN_READONLY|SQLITE_OPEN_URI|SQLITE_OPEN_MAIN_DB || !experimentalSQLAdmissionURI(tls, v, name) {
			return
		}
		b.mainURI.Add(1)
		return experimentalSHMSlotOpen(tls, v, name, file, flags&^SQLITE_OPEN_URI, out)
	}
	// Existing-WAL and unsupported journal routing retain the truthful admission
	// callbacks; WAL never receives the main database/SHM methods table.
	return experimentalSQLAdmissionOpen(tls, v, name, file, flags, out)
}

// Called only after the dedicated connection and pool have closed, with session
// admission stopped. Snapshot registry references without holding its mutex
// while taking file locks (callbacks take file -> registry for generation checks).
func ExperimentalPrivateSQLSessionRetired(v uintptr) error {
	b := experimentalMainReadBindingFor(v)
	if b == nil {
		return fmt.Errorf("missing private SQL binding")
	}
	b.mu.Lock()
	terminal := b.terminal
	b.mu.Unlock()
	if terminal != nil {
		return terminal
	}
	experimentalMainReadMu.Lock()
	var files []*experimentalMainReadFile
	for _, f := range experimentalMainReadFiles {
		if f.binding == b {
			files = append(files, f)
		}
	}
	experimentalMainReadMu.Unlock()
	for _, f := range files {
		f.mu.Lock()
		safe := f.closed && f.result == SQLITE_OK && f.lease == nil
		f.mu.Unlock()
		if !safe {
			return fmt.Errorf("private SQL file not successfully retired")
		}
	}
	return nil
}
