// ExperimentalDuplicateMainFileHandle returns an independently owned,
// non-inheritable metadata-only duplicate of SQLite's actual main handle.
// Call only within sql.Conn.Raw. The caller must close only the duplicate.
func (c *conn) ExperimentalDuplicateMainFileHandle() (uintptr, error) {
	if c == nil {
		return 0, errors.New("sqlite: experimental main identity: nil connection")
	}
	c.Lock()
	defer c.Unlock()
	if c.db == 0 || c.tls == nil || c.inMemory {
		return 0, errors.New("sqlite: experimental main identity: closed or in-memory connection")
	}
	size := int(unsafe.Sizeof(uintptr(0)))
	slot := c.tls.Alloc(size)
	defer c.tls.Free(size)
	*(*uintptr)(unsafe.Pointer(slot)) = 0
	if err := c.fileControl("main", sqlite3.SQLITE_FCNTL_WIN32_GET_HANDLE, slot); err != nil {
		return 0, err
	}
	original := windows.Handle(*(*uintptr)(unsafe.Pointer(slot)))
	if original == 0 || original == windows.InvalidHandle {
		return 0, errors.New("sqlite: experimental main identity: invalid native handle")
	}
	var duplicate windows.Handle
	// Deliberately no DUPLICATE_SAME_ACCESS or DUPLICATE_CLOSE_SOURCE: SQLite's
	// write-capable original remains owned by SQLite. No inherited authority.
	err := windows.DuplicateHandle(windows.CurrentProcess(), original,
		windows.CurrentProcess(), &duplicate,
		windows.FILE_READ_ATTRIBUTES|windows.SYNCHRONIZE, false, 0)
	if err != nil {
		if duplicate != 0 && duplicate != windows.InvalidHandle {
			err = errors.Join(err, windows.CloseHandle(duplicate))
		}
		return 0, err
	}
	if duplicate == 0 || duplicate == windows.InvalidHandle {
		return 0, errors.New("sqlite: experimental main identity: invalid duplicate")
	}
	return uintptr(duplicate), nil
}

// ExperimentalDuplicateMainAndJournalFileHandles observes the actual main and
// currently open journal of this controlled private-cache connection. Current
// WAL is the qualification target, not SHM or complete storage protection.
// Borrowed sqlite3_file/method pointers never escape the locked window.
func (c *experimentalPrivateConn) ExperimentalDuplicateMainAndJournalFileHandles() (result []uintptr, err error) {
	if c == nil || c.conn == nil {
		return nil, errors.New("sqlite: experimental journal identity: nil private connection")
	}
	c.Lock()
	defer c.Unlock()
	if c.db == 0 || c.tls == nil || c.inMemory {
		return nil, errors.New("sqlite: experimental journal identity: closed or in-memory connection")
	}
	mutex := sqlite3.Xsqlite3_db_mutex(c.tls, c.db)
	if mutex == 0 {
		return nil, errors.New("sqlite: experimental journal identity: missing database mutex")
	}
	sqlite3.Xsqlite3_mutex_enter(c.tls, mutex)
	defer sqlite3.Xsqlite3_mutex_leave(c.tls, mutex)
	// FULLMUTEX is recursive: fileControl may enter it again. c.Lock alone only
	// guards driver lifetime; the factory's cache=private excludes shared Pager.
	var owned []windows.Handle
	defer func() {
		if err != nil {
			for i := len(owned) - 1; i >= 0; i-- {
				err = errors.Join(err, windows.CloseHandle(owned[i]))
			}
			result = nil
		}
	}()
	duplicate := func(original windows.Handle) error {
		if original == 0 || original == windows.InvalidHandle {
			return errors.New("sqlite: experimental journal identity: invalid native handle")
		}
		var h windows.Handle
		e := windows.DuplicateHandle(windows.CurrentProcess(), original,
			windows.CurrentProcess(), &h, windows.FILE_READ_ATTRIBUTES|windows.SYNCHRONIZE, false, 0)
		// Own any returned duplicate even on failure, so all error paths clean up.
		if h != 0 && h != windows.InvalidHandle {
			owned = append(owned, h)
		}
		if e != nil {
			return e
		}
		if h == 0 || h == windows.InvalidHandle {
			return errors.New("sqlite: experimental journal identity: invalid duplicate")
		}
		return nil
	}
	size := int(unsafe.Sizeof(uintptr(0)))
	mainSlot := c.tls.Alloc(size)
	defer c.tls.Free(size)
	*(*uintptr)(unsafe.Pointer(mainSlot)) = 0
	if err = c.fileControl("main", sqlite3.SQLITE_FCNTL_WIN32_GET_HANDLE, mainSlot); err != nil {
		return nil, err
	}
	if err = duplicate(windows.Handle(*(*uintptr)(unsafe.Pointer(mainSlot)))); err != nil {
		return nil, err
	}

	journalSlot := c.tls.Alloc(size)
	defer c.tls.Free(size)
	*(*uintptr)(unsafe.Pointer(journalSlot)) = 0
	if err = c.fileControl("main", sqlite3.SQLITE_FCNTL_JOURNAL_POINTER, journalSlot); err != nil {
		return nil, err
	}
	file := *(*uintptr)(unsafe.Pointer(journalSlot))
	if file == 0 {
		return nil, errors.New("sqlite: experimental journal identity: no current journal object")
	}
	methods := (*sqlite3.Tsqlite3_file)(unsafe.Pointer(file)).FpMethods
	if methods == 0 {
		return nil, errors.New("sqlite: experimental journal identity: journal is not open")
	}
	table := (*sqlite3.Tsqlite3_io_methods)(unsafe.Pointer(methods))
	if table.FiVersion < 1 || table.FxFileControl == 0 {
		return nil, errors.New("sqlite: experimental journal identity: unsupported journal methods")
	}
	nativeSlot := c.tls.Alloc(size)
	defer c.tls.Free(size)
	*(*uintptr)(unsafe.Pointer(nativeSlot)) = 0
	// These fields are the public sqlite3_file/sqlite3_io_methods prefix, NOT
	// winFile layout. The thunk representation/signature is fixed to modernc
	// v1.58.0 generated _sqlite3OsFileControl, not a portable Go or Windows ABI.
	call := *(*func(*libc.TLS, uintptr, int32, uintptr) int32)(unsafe.Pointer(&struct{ uintptr }{table.FxFileControl}))
	rc := call(c.tls, file, int32(sqlite3.SQLITE_FCNTL_WIN32_GET_HANDLE), nativeSlot)
	if rc != sqlite3.SQLITE_OK {
		return nil, fmt.Errorf("sqlite: experimental journal WIN32_GET_HANDLE rc=%d", rc)
	}
	if err = duplicate(windows.Handle(*(*uintptr)(unsafe.Pointer(nativeSlot)))); err != nil {
		return nil, err
	}
	return []uintptr{uintptr(owned[0]), uintptr(owned[1])}, nil
}

// Narrow experimental factory: deliberately bypasses Driver.Open's registered
// UDFs, collations, connection hooks and vtabs. It is not a production substitute.
// Only this construction path exposes the journal interface; no global stamps.
type experimentalPrivateDriver struct{}
type experimentalPrivateConn struct{ *conn }

func init() { sql.Register("sqlite-nativeprobe-private", &experimentalPrivateDriver{}) }
func (d *experimentalPrivateDriver) Open(name string) (driver.Conn, error) {
	if err := d.ExperimentalValidateFilename(name); err != nil {
		return nil, err
	}
	// A URL Path, not Opaque or manual percent processing: '%' remains literal
	// filename data. User query/fragment/URI input was rejected before this point.
	uri := (&url.URL{
		Scheme:   "file",
		Path:     "/" + strings.ReplaceAll(name, `\`, "/"),
		RawQuery: url.Values{"cache": {"private"}, "vfs": {"win32"}}.Encode(),
	}).String()
	// Retain modernc's withOpenGate, FULLMUTEX and all base newConn setup.
	c, err := newConn(uri)
	if err != nil {
		return nil, err
	}
	if c == nil {
		return nil, errors.New("sqlite: experimental private factory returned nil connection")
	}
	return &experimentalPrivateConn{conn: c}, nil
}

// Pure validation seam: never opens a path, resolves an alias or normalizes an
// invalid name. These are strict DOS filenames, not arbitrary SQLite DSNs.
func (*experimentalPrivateDriver) ExperimentalValidateFilename(name string) error {
	invalid := errors.New("sqlite: experimental private factory requires a strict absolute DOS filename")
	if !utf8.ValidString(name) || len(name) < 4 || len(name) > 131068 {
		return invalid
	}
	drive := name[0]
	if !((drive >= 'A' && drive <= 'Z') || (drive >= 'a' && drive <= 'z')) || name[1] != ':' || name[2] != 0x5c {
		return invalid
	}
	if len(utf16.Encode([]rune(name))) > 32767 {
		return invalid
	}
	for _, part := range strings.Split(name[3:], `\`) {
		if part == "" || part == "." || part == ".." || strings.HasSuffix(part, ".") || strings.HasSuffix(part, " ") {
			return invalid
		}
		if len(utf16.Encode([]rune(part))) > 255 {
			return invalid
		}
		for _, ch := range part {
			if ch < 0x20 || ch == 0x7f || strings.ContainsRune(`<>:"/|?*#`, ch) {
				return invalid
			}
		}
		base, _, _ := strings.Cut(part, ".")
		base = strings.ToUpper(strings.TrimRight(base, " ."))
		switch base {
		case "CON", "PRN", "AUX", "NUL", "CONIN$", "CONOUT$", "CLOCK$":
			return invalid
		}
		for _, prefix := range []string{"COM", "LPT"} {
			if strings.HasPrefix(base, prefix) {
				suffix := []rune(strings.TrimPrefix(base, prefix))
				if len(suffix) == 1 && ((suffix[0] >= '1' && suffix[0] <= '9') || suffix[0] == '¹' || suffix[0] == '²' || suffix[0] == '³') {
					return invalid
				}
			}
		}
	}
	return nil
}

// Test-binary-only global control. The dedicated child process exits after its
// behavioral proof; the parent must never change its SQLite shared-cache mode.
func (*experimentalPrivateDriver) ExperimentalEnableSharedCacheForSubprocess() error {
	if os.Getenv("SQLITE_PRIVATE_CACHE_CHILD") != "qualification-v1" {
		return errors.New("sqlite: shared-cache control requires dedicated child")
	}
	tls := libc.NewTLS()
	defer tls.Close()
	if rc := sqlite3.Xsqlite3_enable_shared_cache(tls, 1); rc != sqlite3.SQLITE_OK {
		return fmt.Errorf("sqlite: experimental child enable_shared_cache rc=%d", rc)
	}
	return nil
}
